  <script>

    let totalBudget = 0;

    let taxRate = 0.07; // Default 7%

    let items = [];

    let html5QrCode = null;

 

    window.onload = function() {

      loadSavedTrips();

    };

 

    function applySettings() {

      const budgetInput = parseFloat(document.getElementById('initial-budget').value);

      const taxInput = parseFloat(document.getElementById('tax-rate').value);

 

      if (!isNaN(budgetInput) && budgetInput >= 0) {

        totalBudget = budgetInput;

      }

     

      if (!isNaN(taxInput) && taxInput >= 0) {

        taxRate = taxInput / 100;

      }

 

      updateCalculations();

    }

 

    function addItem() {

      const nameInput = document.getElementById('item-name');

      const priceInput = document.getElementById('item-price');

      const categoryInput = document.getElementById('item-category');

      const isTaxable = document.getElementById('item-taxable').checked;

 

      const name = nameInput.value.trim() || 'Item';

      const category = categoryInput.value;

      const price = parseFloat(priceInput.value);

 

      if (isNaN(price) || price <= 0) return;

 

      items.push({

        id: Date.now(),

        name,

        category,

        price,

        isTaxable

      });

 

      nameInput.value = '';

      priceInput.value = '';

      document.getElementById('item-taxable').checked = true;

     

      updateCalculations();

    }

 

    function removeItem(id) {

      items = items.filter(item => item.id !== id);

      updateCalculations();

    }

 

    function clearCurrentCart() {

      if (confirm("Clear all active items from your cart?")) {

        items = [];

        updateCalculations();

      }

    }

 

    function updateCalculations() {

      let subtotal = 0;

      let totalTax = 0;

 

      items.forEach(item => {

        subtotal += item.price;

        if (item.isTaxable) {

          totalTax += (item.price * taxRate);

        }

      });

 

      const grandTotal = subtotal + totalTax;

      const remainingBalance = totalBudget - grandTotal;

 

      document.getElementById('subtotal-val').textContent = `$${subtotal.toFixed(2)}`;

      document.getElementById('tax-val').textContent = `$${totalTax.toFixed(2)}`;

 

      const balanceElement = document.getElementById('remaining-amount');

      balanceElement.textContent = `$${remainingBalance.toFixed(2)}`;

 

      if (remainingBalance < 0) {

        balanceElement.className = 'negative';

      } else {

        balanceElement.className = 'positive';

      }

 

      renderList();

    }

 

    function renderList() {

      const listElement = document.getElementById('item-list');

      listElement.innerHTML = '';

 

      if (items.length === 0) {

        listElement.innerHTML = '<li style="color:#777; font-size:0.9rem;">Cart is empty.</li>';

        return;

      }

 

      items.forEach(item => {

        const itemTax = item.isTaxable ? (item.price * taxRate) : 0;

        const itemTotal = item.price + itemTax;

 

        const li = document.createElement('li');

        li.innerHTML = `

          <div class="item-info">

            <strong>${item.name}</strong>

            <div><span class="category-tag">${item.category}</span></div>

            <span class="item-subtext">$${item.price.toFixed(2)} ${item.isTaxable ? `(+ $${itemTax.toFixed(2)} tax)` : '(No tax)'}</span>

          </div>

          <div style="display:flex; align-items:center; gap: 8px;">

            <b>$${itemTotal.toFixed(2)}</b>

            <button class="remove-btn" onclick="removeItem(${item.id})">✕</button>

          </div>

        `;

        listElement.appendChild(li);

      });

    }

 

    // BARCODE SCANNER LOGIC

    function toggleScanner() {

      const container = document.getElementById('scanner-container');

      if (container.style.display === 'block') {

        stopScanner();

      } else {

        container.style.display = 'block';

        html5QrCode = new Html5Qrcode("interactive");

        html5QrCode.start(

          { facingMode: "environment" },

          { fps: 10, qrbox: { width: 250, height: 150 } },

          (decodedText) => {

            document.getElementById('item-name').value = `Barcode: ${decodedText}`;

            stopScanner();

          },

          () => {} // ignored scan errors

        ).catch(err => {

          alert("Camera access failed or unavailable.");

          stopScanner();

        });

      }

    }

 

    function stopScanner() {

      const container = document.getElementById('scanner-container');

      if (html5QrCode) {

        html5QrCode.stop().then(() => {

          container.style.display = 'none';

        }).catch(() => {

          container.style.display = 'none';

        });

      } else {

        container.style.display = 'none';

      }

    }

 

    // EXPORT & SHARE LOGIC

    function exportCSV() {

      if (items.length === 0) return alert("Cart is empty!");

      let csvContent = "data:text/csv;charset=utf-8,Item,Category,Price,Taxable,Total Cost\n";

     

      items.forEach(i => {

        const itemTax = i.isTaxable ? (i.price * taxRate) : 0;

        csvContent += `"${i.name}","${i.category}",${i.price.toFixed(2)},${i.isTaxable ? 'Yes' : 'No'},${(i.price + itemTax).toFixed(2)}\n`;

      });

 

      const encodedUri = encodeURI(csvContent);

      const link = document.createElement("a");

      link.setAttribute("href", encodedUri);

      link.setAttribute("download", `shopping_list_${new Date().toISOString().slice(0,10)}.csv`);

      document.body.appendChild(link);

      link.click();

      document.body.removeChild(link);

    }

 

    function shareCartText() {

      if (items.length === 0) return alert("Cart is empty!");

      let text = `🛒 *Shopping List*\n`;

      let subtotal = 0;

      let totalTax = 0;

 

      items.forEach(i => {

        const itemTax = i.isTaxable ? (i.price * taxRate) : 0;

        text += `- ${i.name} (${i.category}): $${(i.price + itemTax).toFixed(2)}\n`;

        subtotal += i.price;

        totalTax += itemTax;

      });

 

      const total = subtotal + totalTax;

      text += `\n*Subtotal:* $${subtotal.toFixed(2)}\n*Tax:* $${totalTax.toFixed(2)}\n*Total:* $${total.toFixed(2)}`;

 

      if (navigator.share) {

        navigator.share({ title: 'Shopping List', text: text });

      } else {

        navigator.clipboard.writeText(text);

        alert("Summary copied to clipboard!");

      }

    }

 

    // LOCAL STORAGE FOR SAVED TRIPS

    function saveCurrentTrip() {

      if (items.length === 0) return alert("Add some items before saving a list.");

 

      const listName = prompt("Enter a name for this shopping list:", `Trip ${new Date().toLocaleDateString()}`);

      if (!listName) return;

 

      const savedTrips = JSON.parse(localStorage.getItem('budget_saved_trips') || '[]');

      const subtotal = items.reduce((acc, i) => acc + i.price, 0);

      const tax = items.reduce((acc, i) => acc + (i.isTaxable ? i.price * taxRate : 0), 0);

 

      savedTrips.push({

        id: Date.now(),

        name: listName,

        date: new Date().toLocaleDateString(),

        budget: totalBudget,

        taxRate: taxRate,

        items: items,

        totalSpent: subtotal + tax

      });

 

      localStorage.setItem('budget_saved_trips', JSON.stringify(savedTrips));

      loadSavedTrips();

      alert("List saved!");

    }

 

    function loadSavedTrips() {

      const container = document.getElementById('saved-trips-container');

      const savedTrips = JSON.parse(localStorage.getItem('budget_saved_trips') || '[]');

 

      if (savedTrips.length === 0) {

        container.innerHTML = '<p style="color: #777; font-size: 0.9rem;">No saved lists yet.</p>';

        return;

      }

 

      container.innerHTML = '';

      savedTrips.forEach(trip => {

        const div = document.createElement('div');

        div.className = 'saved-trip-item';

        div.innerHTML = `

          <div class="saved-trip-header">

            <span>${trip.name} (${trip.date})</span>

            <span>$${trip.totalSpent.toFixed(2)}</span>

          </div>

          <div style="font-size: 0.8rem; color: #555; margin-bottom: 6px;">

            ${trip.items.length} items | Budget: $${trip.budget.toFixed(2)}

          </div>

          <div style="display: flex; gap: 6px;">

            <button class="outline small" onclick="reloadTrip(${trip.id})">Load</button>

            <button class="remove-btn small" onclick="deleteTrip(${trip.id})">Delete</button>

          </div>

        `;

        container.appendChild(div);

      });

    }

 

    function reloadTrip(id) {

      const savedTrips = JSON.parse(localStorage.getItem('budget_saved_trips') || '[]');

      const trip = savedTrips.find(t => t.id === id);

     

      if (trip) {

        totalBudget = trip.budget;

        taxRate = trip.taxRate || 0.07;

        items = [...trip.items];

 

        document.getElementById('initial-budget').value = totalBudget;

        document.getElementById('tax-rate').value = (taxRate * 100).toFixed(2);

 

        updateCalculations();

        window.scrollTo({ top: 0, behavior: 'smooth' });

      }

    }

 

    function deleteTrip(id) {

      let savedTrips = JSON.parse(localStorage.getItem('budget_saved_trips') || '[]');

      savedTrips = savedTrips.filter(t => t.id !== id);

      localStorage.setItem('budget_saved_trips', JSON.stringify(savedTrips));

      loadSavedTrips();

    }
